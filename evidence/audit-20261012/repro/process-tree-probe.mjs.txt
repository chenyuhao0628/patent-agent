import fs from 'node:fs/promises';
import {spawn} from 'node:child_process';
import {killManagedProcess} from '/Users/cyh/Documents/ChatGPT/patent-agent/desktop/agent-runner.ts';
const pidFile=new URL('./synthetic-child-pid.txt',import.meta.url).pathname;
const script=`const fs=require('node:fs');const {spawn}=require('node:child_process');const c=spawn(process.execPath,['-e','setInterval(()=>{},1000)'],{stdio:'ignore'});fs.writeFileSync(${JSON.stringify(pidFile)},String(c.pid));c.unref();process.exit(17);`;
const parent=spawn(process.execPath,['-e',script],{detached:true,stdio:'ignore'});
await new Promise((r,j)=>{parent.once('exit',r);parent.once('error',j);});
const pid=Number(await fs.readFile(pidFile,'utf8'));
let before=false,after=false;
try{process.kill(pid,0);before=true;}catch{}
await killManagedProcess(parent);
try{process.kill(pid,0);after=true;}catch{}
const result={platform:process.platform,parentExitCode:parent.exitCode,ownedChildAliveBeforeCleanup:before,ownedChildAliveAfterCleanup:after};
// Only the synthetic process started by this probe is terminated here.
if(after)try{process.kill(pid,'SIGKILL');}catch{}
await fs.writeFile(new URL('./process-tree-probe.json',import.meta.url),JSON.stringify(result,null,2));
console.log(JSON.stringify(result,null,2));
